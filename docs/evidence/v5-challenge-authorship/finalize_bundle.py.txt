"""Freeze the private challenge and verify its final bytes without target-model execution."""
import hashlib
import json
import os
from pathlib import Path
from typing import Any

ROOT = Path(__file__).resolve().parent
REPO = ROOT.parents[1]


def sha(path: Path) -> str:
    return hashlib.sha256(path.read_bytes()).hexdigest()


def save(name: str, value: Any) -> None:
    with (ROOT / name).open("x") as stream:
        stream.write(json.dumps(value, indent=2, sort_keys=True) + "\n")


def main() -> None:

    manifest = json.loads(manifest_path.read_text())
    inventory = json.loads((ROOT / "input-inventory.json").read_text())
    oracles = json.loads((ROOT / "manual-oracles.json").read_text())
    validation = json.loads((ROOT / "validation.json").read_text())
    checks = json.loads((ROOT / "strict-typecheck.json").read_text())
    assert checks["summary"]["errorCount"] == checks["summary"]["warningCount"] == 0
    assert validation["status"] == "passed"
    assert len(manifest["cases"]) == len(oracles["cases"]) == 30
    for row in manifest["cases"]:
        assert sha(ROOT / row["scenario_file"]) == row["scenario_hash"]
    for item in inventory["inventory"]:
        assert sha(REPO / item["path"]) == item["file_sha256"]
    for item in inventory["manifest_as_of_inventory"]:
        assert sha(REPO / item["path"]) == item["file_sha256"]

        assert sha(ROOT / "pinned-reference" / item["path"]) == item["sha256"]

        "oracle_order": ["Save original literals and AI judgment labels", "Run independent civil-calendar integer UTC oracle", "Record and resolve static runner input compatibility issue, preserving original bytes", "Save revised judgment labels and complete independent exact temporal oracles", "Inventory prior inputs without labels", "Import only pinned typed contracts and policy; crosscheck every case", "Validate manifest, cardinalities, hashes and strict types", "Freeze all bundle files with SHA256SUMS"],
        "temporal_method": "No floating-point epoch timestamps. Independent integer Gregorian days and offsets produce exact UTC microseconds, age, threshold delta, and event-to-last/cutoff relations. Pydantic/datetime verification happens only after these outputs are saved.",
        "coverage": "See coverage.json. All four order states and four shipment states, threshold equality and microseconds on either side, positive and negative fractional offsets, missing and naive carrier timestamps, future timestamps, chronology contradictions, explicit conflict flags, mixed-case terminal event codes, empty and unsupported exceptions, unordered valid history, and three untrusted hostile-text cases are represented.",
        "fault_contract": {"expire": 2, "order_version": 2, "carrier_version": 2, "ticket_drop": 3, "annotation_drop": 3, "decision_replay": 3, "none": 15},
        "denominators": {"total": 30, "stale": 6, "authorized_completion": 24, "hostile": 3},
