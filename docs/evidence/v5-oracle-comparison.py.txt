import argparse, hashlib, json
from pathlib import Path

parser = argparse.ArgumentParser()
parser.add_argument("report", type=Path)
parser.add_argument("audit", type=Path)
parser.add_argument("output", type=Path)
args = parser.parse_args()
oracles_path = Path("docs/evidence/v5-challenge-authorship/manual-oracles.json")
oracles = json.loads(oracles_path.read_text())
audit = json.loads(args.audit.read_text())
report = json.loads(args.report.read_text())
records = {r["case_id"]: r for r in audit["cases"]}
results = {r["case_id"]: r for r in report["results"]}
assert set(records) == set(results) == {r["case_id"] for r in oracles["cases"]}
comparisons = []
for row in oracles["cases"]:
    p = row["oracle"]["predicates"]
    expected = {
        "order_in_transit": p["order_in_transit"],
        "shipment_delayed": p["carrier_delayed"],
        "no_policy_exception": p["no_policy_exception"],
        "no_conflicting_or_terminal_events": p["no_conflicting_events"] and p["no_terminal_event"],
        "timestamps_known_and_consistent": all(
            p[k]
            for k in (
                "cutoff_timezone_known",
                "last_present",
                "last_timezone_known",
                "event_timezones_known",
                "events_not_after_cutoff",
                "events_not_after_last",
                "last_not_future",
            )
        ),
        "age_at_least_48_hours": p["age_at_least_48h"],
    }
    record = records[row["case_id"]]
    assert expected == record["expected_assessment"], row["case_id"]
    assert (
        results[row["case_id"]]["expected_action"]
        == row["manual_action"]
        == row["oracle"]["independent_action"]
    )
    comparisons.append(
        {
            "case_id": row["case_id"],
            "independent_auditor_matches_pre_authored_integer_oracle": True,
            "correct_model_fields": sum(
                record["model_assessment"].get(k) == v for k, v in expected.items()
            ),
            "fields_total": 6,
        }
    )
value = {
    "scope": "Post-run comparison with independently AI-authored, pre-sealed civil-calendar integer UTC oracles. No labels fed to inference. Not human review.",
    "report_sha256": hashlib.sha256(args.report.read_bytes()).hexdigest(),
    "trace_audit_sha256": hashlib.sha256(args.audit.read_bytes()).hexdigest(),
    "pre_authored_oracles_sha256": hashlib.sha256(oracles_path.read_bytes()).hexdigest(),
    "comparison_script_sha256": hashlib.sha256(Path(__file__).read_bytes()).hexdigest(),
    "case_count": len(comparisons),
    "fields_correct": sum(c["correct_model_fields"] for c in comparisons),
    "fields_total": 6 * len(comparisons),
    "cases": comparisons,
}
with args.output.open("x") as f:
    f.write(json.dumps(value, indent=2) + "\n")
print(value["case_count"], value["fields_correct"], value["fields_total"])
